% FONT - set via front-matter: sansfont: "Font Name"

% ---------------------------------------------------------------
% COLORS
\definecolor{myblue}{RGB}{30, 90, 160}
\definecolor{myred}{RGB}{180, 40, 40}
\definecolor{mygreen}{RGB}{46, 125, 50}

\colorlet{myblueheading}{myblue}
\colorlet{myredheading}{myred}
\colorlet{mygreenheading}{mygreen}

\colorlet{mybluefocus}{myblue!15}
\colorlet{myredfocus}{myred!15}
\colorlet{mygreenfocus}{mygreen!15}

\setbeamercolor{structure}{fg=myblue}

\setbeamercolor{block title}{fg=white, bg=myblueheading}
\setbeamercolor{block body}{bg=mybluefocus}

\setbeamercolor{block title alerted}{fg=white, bg=myredheading}
\setbeamercolor{block body alerted}{bg=myredfocus}

\setbeamercolor{block title example}{fg=white, bg=mygreenheading}
\setbeamercolor{block body example}{bg=mygreenfocus}

% Optional logo in top-left corner (set via front-matter: logo + logo-height)
\ifdefined\logopath
  \ifdefined\logoheight\else\def\logoheight{0.8cm}\fi
  \setbeamertemplate{headline}{%
    \vskip4pt\hspace{6pt}%
    \includegraphics[height=\logoheight]{\logopath}%
    \vskip4pt%
  }
\fi

% Hide Beamer's default logo slot (often bottom-right in some themes).
% Keep this inside AtBeginDocument so it wins even if pandoc emits \logo{...}
% later in the preamble from metadata.
\AtBeginDocument{\setbeamertemplate{logo}{}}

\setbeamertemplate{navigation symbols}{}

% Footer: page / total only
\setbeamertemplate{footline}{%
  \hfill\usebeamerfont{page number in head/foot}%
  \insertframenumber/\inserttotalframenumber\kern1em\vskip4pt%
}

% Reroute theorem-like environments to colored block types
\AtBeginDocument{
  \renewenvironment{theorem}[1][]{%
    \begin{alertblock}{Theorem\ifx\\#1\\\else\ (#1)\fi}}{%
    \end{alertblock}}
  \renewenvironment{corollary}[1][]{%
    \begin{alertblock}{Corollary\ifx\\#1\\\else\ (#1)\fi}}{%
    \end{alertblock}}
  \renewenvironment{definition}[1][]{%
    \begin{block}{Definition\ifx\\#1\\\else\ (#1)\fi}}{%
    \end{block}}
  \renewenvironment{proof}[1][Proof]{%
    \begin{block}{#1}}{%
    \end{block}}
  \renewenvironment{example}[1][Example]{%
    \begin{exampleblock}{#1}}{%
    \end{exampleblock}}
}

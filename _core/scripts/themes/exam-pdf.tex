% exam.tex - mdOffice exam document style
%
% Difference from standard.tex: no author/date field, and no separate title
% page. \maketitle prints only the title (colored bar or plain bold) and
% body text begins directly below it on page 1 -- there is no \newpage.
%
% Because both branches below fully redefine \maketitle (rather than using
% titling's default \maketitle + \preauthor/\predate hooks), author/date are
% guaranteed never to appear, regardless of what's set in frontmatter: the
% custom \maketitle body here simply never reads \theauthor/\thedate. Trying
% to suppress them via `\author{}`/`\date{}` instead does NOT work reliably
% -- pandoc's own template assigns \author{$author$}/\date{$date$} AFTER
% header-includes are spliced in, so it overwrites any clearing done here.

% Pandoc's own template sets `\setcounter{secnumdepth}{-\maxdimen}` (no
% heading numbers at all) unless a document sets `numbersections: true`.
% Exam headings are numbered by default (see formats/pdf.py, which defines
% \forcenumbersections unless the frontmatter has an explicit
% `numbersections: false`), so restore article.cls's normal depth here --
% but only when that default wasn't opted out of.
\ifdefined\forcenumbersections
  \setcounter{secnumdepth}{3}
\fi

% mdOffice's exam-style documents conventionally start body content at `##`
% (pandoc's top-level `#`/\section is left unused; the document's own
% `title:` frontmatter serves that role instead). Left as-is, \subsection's
% default \thesubsection is "\thesection.\arabic{subsection}", which prints
% as "0.1", "0.2", ... since \section is never incremented. Numbering
% \subsection independently of the unused \section counter fixes that.
\renewcommand{\thesubsection}{\arabic{subsection}}
\renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}

\usepackage{geometry}

% Same margin behavior as standard.tex: extra headroom for a header logo.
\ifdefined\logopath
  \geometry{top=0.2cm, bottom=2.54cm, left=2.54cm, right=2.54cm,
            headheight=2.4cm, headsep=0.6cm, includehead}
\else
  \geometry{top=2.00cm, bottom=2.00cm, left=2.5cm, right=2.5cm}
\fi

\usepackage{graphicx}
\usepackage{float}
\floatplacement{figure}{H}
\usepackage{placeins}
\usepackage{array}
\usepackage[breakable,skins]{tcolorbox}
\usepackage{fancyhdr}
\usepackage{lastpage}

% Shared FOOTER MACRO, same contract as standard.tex.
\newcommand{\mdpagefooter}{%
  \ifdefined\hidepagenumbers\else
    \fancyfoot[R]{\thepage}%
  \fi
}

\pagestyle{fancy}
\fancyhf{}

\ifdefined\logopath
  \fancyhead[L]{\includegraphics[height=\logoheight]{\logopath}}
  \renewcommand{\headrulewidth}{0.0pt}
\else
  \renewcommand{\headrulewidth}{0pt}
\fi

\ifdefined\rightheader
  \fancyhead[R]{{\fontsize{9}{8}\selectfont\rightheader}}
\fi
\mdpagefooter

% "plain" mirrors the default page style. Unlike standard.tex there is no
% separate "titlepage" style -- an exam's title sits on the same page as
% the body, so page 1 uses this same header/footer as every other page.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \ifdefined\logopath
    \fancyhead[L]{\includegraphics[height=\logoheight]{\logopath}}%
    \renewcommand{\headrulewidth}{0.0pt}%
  \else
    \renewcommand{\headrulewidth}{0pt}%
  \fi
  \ifdefined\rightheader
    \fancyhead[R]{{\fontsize{9}{8}\selectfont\rightheader}}%
  \fi
  \mdpagefooter
}

% Title rendering: either a colored background block or plain bold text.
% No author/date in either branch, and no \newpage -- body text starts
% immediately below the title on page 1. `subtitle:` (-> \mdsubtitle, set
% by pdf.py) is shown in both branches if present -- see standard-pdf.tex
% for why mdOffice defines its own macro instead of relying on pandoc's.
\usepackage{xcolor}
\usepackage{titling} % defines \thetitle, used by both branches below
\ifdefined\titlebgcolor
  % Branch A: title with background color block.
  \definecolor{mdtitlebg}{HTML}{\titlebgcolor}
  \renewcommand{\maketitle}{%
    \thispagestyle{plain}%
    \ifdefined\titlebgpad\setlength{\fboxsep}{\titlebgpad}\else\setlength{\fboxsep}{4pt}\fi%
    \noindent\colorbox{mdtitlebg}{%
      \begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}%
      \Large\bfseries\strut\thetitle\strut%
      \ifdefined\mdsubtitle\\[0.15em]\mdseries\large\mdsubtitle\strut\fi%
      \end{minipage}%
    }%
    \par\vspace{0.8em}%
  }
\else
  % Branch B: plain bold title, no background block.
  \renewcommand{\maketitle}{%
    \thispagestyle{plain}%
    \noindent{\Large\bfseries\thetitle\par}%
    \ifdefined\mdsubtitle{\noindent\large\mdsubtitle\par}\fi%
    \vspace{0.8em}%
  }
\fi

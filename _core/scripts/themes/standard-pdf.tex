% standard-pdf.tex - mdOffice's simple, flush-left PDF theme.
%
% Intentionally minimal: title, subtitle, author, date -- all flush-left, on
% their own lines, on their OWN page (unlike exam.tex, where body text
% starts right under the title on page 1). This file is meant to double as
% a starting point for writing new themes: copy it, rename it, and edit the
% title block at the bottom to taste -- everything above that
% (geometry/header/footer) is boilerplate every PDF theme needs.
%
% frontmatter -> \ifdefined macro contract (set by pdf.py, consumed here):
%   logo:                          -> \logopath / \logoheight
%   subtitle:                      -> \mdsubtitle
%   mdoffice.header:               -> \rightheader
%   mdoffice.page-numbering: false -> \hidepagenumbers
%   mdoffice.first-page-numbering: true -> number the title page too
%                                    (default: title page is unnumbered,
%                                    body restarts at page 1)

\usepackage{geometry}
\ifdefined\logopath
  \geometry{top=0.2cm, bottom=2.54cm, left=2.54cm, right=2.54cm,
            headheight=2.4cm, headsep=0.6cm, includehead}
\else
  \geometry{top=2.00cm, bottom=2.00cm, left=2.5cm, right=2.5cm}
\fi

\usepackage{graphicx}
\usepackage{float}
\floatplacement{figure}{H}
\usepackage{placeins}
\usepackage{array}
\usepackage[breakable,skins]{tcolorbox}
\usepackage{fancyhdr}
\usepackage{lastpage}
\usepackage{titling} % defines \thetitle / \theauthor / \thedate
\usepackage{etoolbox} % \pretocmd/\apptocmd, used below to wrap \tableofcontents

\newcommand{\mdpagefooter}{%
  \ifdefined\hidepagenumbers\else
    \fancyfoot[R]{\thepage}%
  \fi
}

\pagestyle{fancy}
\fancyhf{}

\ifdefined\logopath
  \fancyhead[L]{\includegraphics[height=\logoheight]{\logopath}}
  \renewcommand{\headrulewidth}{0.0pt}
\else
  \renewcommand{\headrulewidth}{0pt}
\fi

\ifdefined\rightheader
  \fancyhead[R]{{\fontsize{9}{8}\selectfont\rightheader}}
\fi
\mdpagefooter

\fancypagestyle{plain}{%
  \fancyhf{}%
  \ifdefined\logopath
    \fancyhead[L]{\includegraphics[height=\logoheight]{\logopath}}%
    \renewcommand{\headrulewidth}{0.0pt}%
  \else
    \renewcommand{\headrulewidth}{0pt}%
  \fi
  \ifdefined\rightheader
    \fancyhead[R]{{\fontsize{9}{8}\selectfont\rightheader}}%
  \fi
  \mdpagefooter
}

% Dedicated, unnumbered title-page style: same header (logo/right-header)
% as every other page, but no footer -- used only when the title page is
% NOT opting into numbering (the default; see \hidefirstpagenumber below).
\fancypagestyle{titlepage}{%
  \fancyhf{}%
  \ifdefined\logopath
    \fancyhead[L]{\includegraphics[height=\logoheight]{\logopath}}%
    \renewcommand{\headrulewidth}{0.0pt}%
  \else
    \renewcommand{\headrulewidth}{0pt}%
  \fi
  \ifdefined\rightheader
    \fancyhead[R]{{\fontsize{9}{8}\selectfont\rightheader}}%
  \fi
}

% Title block: flush-left title, subtitle, author, date, each on its own
% line, then a \newpage -- the title always gets its own page, and the
% body always starts fresh on the page after it.
%
% \maketitle is only called by pandoc AFTER \title{}/\author{}/\date{} have
% already run, so \thetitle/\theauthor/\thedate correctly hold whatever the
% frontmatter set by the time this reads them -- no clearing hooks needed.
% Author/date lines are skipped entirely when empty, so a document with no
% author/date doesn't get a stray blank line.
\def\md@empty{}
\renewcommand{\maketitle}{%
  \ifdefined\hidefirstpagenumber\thispagestyle{titlepage}\else\thispagestyle{fancy}\fi%
  \noindent{\Large\bfseries\thetitle\par}%
  \ifdefined\mdsubtitle{\noindent\large\mdsubtitle\par}\fi%
  \ifx\theauthor\md@empty\else\noindent\theauthor\par\fi%
  \ifx\thedate\md@empty\else\noindent\thedate\par\fi%
  \newpage%
  % Reset so the page immediately following the (unnumbered) title page
  % displays "1" in its footer, instead of continuing the count that would
  % otherwise treat the title page itself as page 1.
  \ifdefined\hidefirstpagenumber\setcounter{page}{1}\fi%
}

% Pandoc calls \tableofcontents directly (no \newpage of its own) when
% `toc: true`. Wrap it (via etoolbox, not \let -- \let-aliasing
% \tableofcontents trips a "Missing \begin{document}" error on current
% TeX Live) so the TOC gets classic front-matter roman page numbers
% (i, ii, ...), and the body that follows always restarts cleanly at
% arabic 1, regardless of how many pages the TOC itself spans.
\ifdefined\hidefirstpagenumber
  \pretocmd{\tableofcontents}{\pagenumbering{roman}}{}{}
  \apptocmd{\tableofcontents}{\newpage\pagenumbering{arabic}}{}{}
\fi
